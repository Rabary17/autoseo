  • Rendu : 100 % SSG + ISR. Aucun contenu éditorial en CSR. Les 50 derniers articles sont générés au build (generateStaticParams), le reste à la demande puis mis en cache.
  • Fraîcheur : webhook WP publish_post / save_post → POST /api/revalidate → revalidateTag. Statique + quasi temps réel.
  • SEO : Yoast reste la source de vérité. yoast_head_json (natif depuis Yoast 14) fournit title, description, canonical, OG et le graphe JSON-LD complet ; fallback généré côté front si absent (§6).
  • Sitemap : généré par Yoast côté WP, proxifié via rewrites Next.js sous le domaine frontend (§6.4).
  • Sécurité : le CMS vit sur un sous-domaine (cms.monmag.fr) non indexé (réglage Yoast + X-Robots-Tag), seul le frontend est public.

lib/wp.ts — fetch centralisé : timeout, retry exponentiel, cache par tags

import type { WpPost, WpTerm, WpUser } from "./types";

const WP = process.env.WP_API_URL!;  // https://cms.monmag.fr/wp-json
const TIMEOUT_MS = 8_000;

export class WpNotFound extends Error {}

type Opts = { revalidate?: number; tags?: string[]; retries?: number };

async function wpFetch(path: string,
  { revalidate = 600, tags = [], retries = 2 }: Opts = {}) {
  let lastErr: unknown;
  for (let i = 0; i <= retries; i++) {
    try {
      const res = await fetch(WP + path, {
        next: { revalidate, tags },   // cache + tags de revalidation
        signal: AbortSignal.timeout(TIMEOUT_MS),
      });
      if (res.status === 404) throw new WpNotFound(path);
      if (!res.ok) throw new Error(`WP ${res.status} — ${path}`);
      return res;
    } catch (e) {
      if (e instanceof WpNotFound) throw e;    // 404 : pas de retry
      lastErr = e;
      await new Promise((r) => setTimeout(r, 300 * 2 ** i)); // backoff
    }
  }
  throw lastErr;
}

const wpJson = async <T>(path: string, opts?: Opts): Promise<T> =>
  (await wpFetch(path, opts)).json();

/* ---------- Articles ---------- */

export async function getPosts(page = 1, perPage = 12, extra = "") {
  const res = await wpFetch(
    "/wp/v2/posts?_embed=1&status=publish" +
      `&page=${page}&per_page=${perPage}${extra}`,
    { tags: ["posts"] },
  );
  return {
    posts: (await res.json()) as WpPost[],
    total: Number(res.headers.get("X-WP-Total")),
    totalPages: Number(res.headers.get("X-WP-TotalPages")),
  };
}

export async function getPostBySlug(slug: string): Promise<WpPost | null> {
  try {
    const posts = await wpJson<WpPost[]>(
      `/wp/v2/posts?slug=${encodeURIComponent(slug)}` +
        "&_embed=1&status=publish",
      { tags: ["posts", `post:${slug}`] },
    );
    return posts[0] ?? null;          // dépublié / absent → 404
  } catch { return null; }
}

/* ---------- Taxonomies & auteurs ---------- */

export const getCategories = () =>
  wpJson<WpTerm[]>(
    "/wp/v2/categories?per_page=8&orderby=count&hide_empty=true",
    { tags: ["categories"], revalidate: 3600 });

export async function getTermBySlug(tax: "categories" | "tags", slug: string) {
  const terms = await wpJson<WpTerm[]>(
    `/wp/v2/${tax}?slug=${slug}`, { tags: [tax] });
  return terms[0] ?? null;
}

export async function getAuthorBySlug(slug: string) {
  const users = await wpJson<WpUser[]>(
    `/wp/v2/users?slug=${slug}`, { tags: ["users"] });
  return users[0] ?? null;
}

export const getPostsByAuthor = (id: number, page = 1) =>
  getPosts(page, 12, `&author=${id}`);
export const getPostsByCategory = (id: number, page = 1) =>
  getPosts(page, 12, `&categories=${id}`);

4. Revalidation ISR & plugin WordPress

app/api/revalidate/route.ts — endpoint appelé par WordPress

import { revalidateTag } from "next/cache";

export async function POST(req: Request) {
  if (req.headers.get("x-revalidate-secret") !== process.env.REVALIDATE_SECRET) {
    return new Response("Forbidden", { status: 403 });
  }
  const { slug } = await req.json().catch(() => ({}));
  revalidateTag("posts");             // accueil, catégories, auteurs
  if (slug) revalidateTag(`post:${slug}`); // la page article
  return Response.json(
    { revalidated: true, slug: slug ?? "all", at: Date.now() });
}

wp-plugin/headless-bridge.php — à déposer dans wp-content/plugins/ puis activer

<?php
/**
 * Plugin Name: Headless Bridge (Yoast + ISR)
 * Description: Expose yoast_head_json et notifie le frontend
 *              Next.js à chaque publication / mise à jour.
 */

define('HB_FRONT_URL', 'https://www.monmag.fr');
define('HB_SECRET', 'CHANGEZ-MOI');   // = REVALIDATE_SECRET (Next.js)

// yoast_head_json est natif depuis Yoast 14 — on le garde exposé.
add_filter('wpseo_enable_rest_api', '__return_true');

// Webhook de revalidation ISR
function hb_notify_front($post_id, $post = null) {
  $post = $post ?: get_post($post_id);
  if (!$post || wp_is_post_revision($post_id)) return;
  if ($post->post_type !== 'post') return;
  wp_remote_post(HB_FRONT_URL . '/api/revalidate', [
    'blocking' => false,              // ne bloque pas l'admin WP
    'timeout'  => 3,
    'headers'  => [
      'Content-Type'        => 'application/json',
      'x-revalidate-secret' => HB_SECRET,
    ],
    'body' => wp_json_encode(['slug' => $post->post_name]),
  ]);
}
add_action('publish_post',   'hb_notify_front', 10, 2);
add_action('save_post_post', 'hb_notify_front', 10, 2);
add_action('trashed_post',   'hb_notify_front');  // dépublication

// Sitemap Yoast : réécrire le domaine CMS → domaine frontend
$hb_swap_host = fn($url) =>
  str_replace(home_url(), HB_FRONT_URL, $url);
add_filter('wpseo_xml_sitemap_post_url', $hb_swap_host);
add_filter('wpseo_sitemap_url',          $hb_swap_host);

import Image from "next/image";
import Link from "next/link";
import type { WpPost } from "@/lib/types";

export default function ArticleCard({ post, priority = false }:
  { post: WpPost; priority?: boolean }) {
  const media = post._embedded?.["wp:featuredmedia"]?.[0];
  const cat   = post._embedded?.["wp:term"]?.[0]?.[0];
  return (
    <article>
      {media && (
        <Link href={`/${post.slug}`} prefetch tabIndex={-1} aria-hidden>

            src={media.source_url}
            alt={media.alt_text || ""}
            width={media.media_details.width}
            height={media.media_details.height}
            sizes="(max-width: 768px) 100vw, 33vw"
            priority={priority}                 // true pour la une → LCP
          />
        </Link>
      )}
      {cat && (
        <p><Link href={`/categorie/${cat.slug}`}>{cat.name}</Link></p>
      )}
      <h3>
        <Link href={`/${post.slug}`} prefetch
          dangerouslySetInnerHTML={ { __html: post.title.rendered } } />
      </h3>
      <time dateTime={post.date}>
        {new Date(post.date)
          .toLocaleDateString("fr-FR", { dateStyle: "long" })}
      </time>
    </article>
  );
}

components/Breadcrumb.tsx — visible + JSON-LD BreadcrumbList (fallback si Yoast absent)

import Link from "next/link";

type Crumb = { name: string; href: string };

export default function Breadcrumb({ items, withSchema = false }:
  { items: Crumb[]; withSchema?: boolean }) {
  const schema = {
    "@context": "https://schema.org",
    "@type": "BreadcrumbList",
    itemListElement: items.map((c, i) => ({
      "@type": "ListItem", position: i + 1, name: c.name,
      item: process.env.SITE_URL + c.href,
    })),
  };
  return (
    <nav aria-label="Fil d'Ariane">
      <ol>
        {items.map((c, i) => (
          <li key={c.href}>
            {i < items.length - 1
              ? <Link href={c.href} prefetch>{c.name}</Link>
              : <span aria-current="page">{c.name}</span>}
          </li>
        ))}
      </ol>
      {withSchema && (
        <script type="application/ld+json"
          dangerouslySetInnerHTML={ { __html: JSON.stringify(schema) } } />
      )}
    </nav>
  );
}

app/[slug]/page.tsx — la page article complète (metadata Yoast + JSON-LD + TL;DR)

import type { Metadata } from "next";
import { notFound } from "next/navigation";
import Image from "next/image";
import { getPostBySlug, getPosts } from "@/lib/wp";
import Breadcrumb from "@/components/Breadcrumb";
import { fallbackArticleSchema } from "@/lib/seo";

const dateFr = (d: string) =>
  new Date(d).toLocaleDateString("fr-FR", { dateStyle: "long" });

export const revalidate = 600;      // filet de sécurité (+ webhook)
export const dynamicParams = true;  // hors build → génération ISR

export async function generateStaticParams() {
  const { posts } = await getPosts(1, 50);          // 50 derniers au build
  return posts.map((p) => ({ slug: p.slug }));
}

type Props = { params: Promise<{ slug: string }> };

export async function generateMetadata({ params }: Props):
  Promise<Metadata> {
  const post = await getPostBySlug((await params).slug);
  const y = post?.yoast_head_json;
  if (!post) return {};
  return {                            // Yoast = source de vérité
    title: y?.title,
    description: y?.description,
    alternates: { canonical: y?.canonical },
    openGraph: {
      type: "article",
      images: y?.og_image?.map((i) => i.url),
    },
  };
}

export default async function ArticlePage({ params }: Props) {
  const post = await getPostBySlug((await params).slug);
  if (!post) notFound();              // dépublié → 404 gracieux

  const media  = post._embedded?.["wp:featuredmedia"]?.[0];
  const author = post._embedded?.author?.[0];
  const cat    = post._embedded?.["wp:term"]?.[0]?.[0];
  const schema = post.yoast_head_json?.schema  // Yoast tel quel…
    ?? fallbackArticleSchema(post);                 // …sinon fallback front

  return (
    <>
      <Breadcrumb withSchema={!post.yoast_head_json} items={[
        { name: "Accueil", href: "/" },
        ...(cat
          ? [{ name: cat.name, href: `/categorie/${cat.slug}` }]
          : []),
        { name: post.title.rendered, href: `/${post.slug}` },
      ]} />

      <article>
        <header>
          <h1 dangerouslySetInnerHTML={ { __html: post.title.rendered } } />
          <p>
            Par <a href={`/auteur/${author?.slug}`} rel="author">
              {author?.name}</a>
            {" · "}Publié le <time dateTime={post.date}>
              {dateFr(post.date)}
            </time>
            {post.modified !== post.date && (
              <> · Mis à jour le <time dateTime={post.modified}>
                {dateFr(post.modified)}
              </time></>
            )}
          </p>
        </header>

        {post.acf?.tldr && (          /* TL;DR extractible (LLM) */
          <aside aria-label="L'essentiel">
            <h2>L'essentiel</h2><p>{post.acf.tldr}</p>
          </aside>
        )}

        {media && (
          <Image src={media.source_url} alt={media.alt_text || ""}
            width={media.media_details.width}
            height={media.media_details.height}
            sizes="(max-width: 768px) 100vw, 720px"
            priority fetchPriority="high" />      /* LCP */
        )}

        <div
          dangerouslySetInnerHTML={ { __html: post.content.rendered } } />

        {post.acf?.sources?.length ? ( /* EEAT : sources vérifiables */
          <footer>
            <h2>Sources</h2>
            <ul>{post.acf.sources.map((s) => (
              <li key={s.url}>
                <a href={s.url} rel="external noopener">{s.label}</a>
              </li>
            ))}</ul>
          </footer>
        ) : null}
      </article>

      <script type="application/ld+json"
        dangerouslySetInnerHTML={ { __html: JSON.stringify(schema) } } />
    </>
  );
}

app/auteur/[slug]/page.tsx — page auteur, articles filtrés via wp-json

import { notFound } from "next/navigation";

8. Checklist finale de conformité

Core Web Vitals

  • LCP < 2,5 s : image de une en priority + fetchPriority="high"
  • CLS < 0,1 : width/height explicites partout, fonts via next/font (zéro FOUT)
  • INP < 200 ms : Server Components par défaut, JS client minimal
  • AVIF/WebP via next/image, lazy-loading hors viewport
  • Cache edge : pages ISR servies statiques (vérifier x-vercel-cache: HIT)
  • Prefetch des liens internes visibles (next/link par défaut)

SEO technique

  • Title + meta description uniques (Yoast) sur 100 % des pages
  • Canonical systématique = URL frontend (vérifier après proxy)
  • Un seul H1 par page, hiérarchie Hn sans saut
  • Sitemap Yoast accessible sur le domaine frontend + soumis à GSC
  • robots.txt : /api/ et paramètres d'URL exclus
  • CMS (cms.monmag.fr) non indexé : noindex global + X-Robots-Tag
  • 404 réel (status code) pour contenu dépublié

GEO — visibilité IA

  • JSON-LD @graph complet (Article, Person, Organization, BreadcrumbList, WebSite+SearchAction) — validé sur validator.schema.org
  • TL;DR « L'essentiel » en tête d'article (champ ACF)
  • datePublished / dateModified visibles ET balisés, cohérents Yoast
  • llms.txt à la racine, à jour à chaque nouvelle rubrique
  • sameAs cohérents : mêmes profils sur site, Yoast et réseaux
  • GPTBot / ClaudeBot / PerplexityBot non bloqués dans robots.txt

EEAT & UX

  • 7 pages EEAT publiées et liées depuis le footer
  • Chaque article signé → page auteur avec bio + sameAs
  • Bloc « Sources » avec liens externes vérifiables
  • Fil d'Ariane visible sur toutes les pages profondes
  • Contraste WCAG AA, cibles tactiles ≥ 44 px, zéro interstitiel intrusif
  • Pagination crawlable (liens réels /page/2), pas d'infinite scroll seul
  • Maillage interne contextuel : articles liés par catégorie/tag en fin d'article

9. Déploiement multi-sites (N magazines réels)

Un seul dépôt de code, un fichier de configuration par magazine, une instance WordPress distincte par site. Chaque déploiement sélectionne sa configuration via la variable d'environnement SITE_KEY — c'est le paramètre qui change d'un site à l'autre : nom, domaine, source WP, charte graphique, entité éditoriale et profils sociaux propres. Chaque magazine reste une entité réelle et autonome (contenu, équipe, pages EEAT), et non un gabarit dupliqué.

config/sites.ts — registre des sites, résolution du site actif au build

import type { NextConfig } from "next";

/* Un enregistrement par magazine du réseau. La clé = SITE_KEY (env). */
export interface SiteConfig {
  key: string;             // "cuisine" | "voyage" | …
  name: string;            // nom éditorial affiché
  domain: string;          // www.monmag-cuisine.fr
  wpApi: string;           // https://cms-cuisine.mondomaine.fr/wp-json
  locale: string;          // "fr-FR"
  brand: {
    primary: string;       // couleur d'accent du thème
    font: string;          // pairing de polices propre au site
    logo: string;          // /brands/cuisine/logo.svg
  };
  org: {                   // alimente Organization + sameAs (par site)
    legalName: string;
    sameAs: string[];      // réseaux sociaux PROPRES à ce magazine
  };
}

export const SITES: Record<string, SiteConfig> = {
  cuisine: {
    key: "cuisine",
    name: "Le Carnet Gourmand",
    domain: "www.carnet-gourmand.fr",
    wpApi: "https://cms.carnet-gourmand.fr/wp-json",
    locale: "fr-FR",
    brand: { primary: "#B4552D", font: "Fraunces, serif",
             logo: "/brands/cuisine/logo.svg" },
    org: { legalName: "Le Carnet Gourmand SAS",
           sameAs: ["https://instagram.com/carnetgourmand"] },
  },
  // … 49 autres entrées, chacune 100% distincte
};

/* Site actif résolu au build depuis l'env de déploiement. */
export const SITE: SiteConfig = SITES[process.env.SITE_KEY ?? ""] ??
  (() => { throw new Error("SITE_KEY inconnu : " + process.env.SITE_KEY); })();

import { SITE } from "@/config/sites";

}
